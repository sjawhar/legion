// Throwaway INFRA-531 tester probe; never committed.
import { expect, test } from "@playwright/test";

import { createComment, createIssue, createProject, editArtifact, getArtifact, getArtifactText, getArtifactVersion } from "./api";
import { documentEditor, openSpec } from "./editor";
import { resetDatabase } from "./seed";
import { asUser } from "./users";

const session = {
  actor: { kind: "session" as const, id: "e2e-session", origin: { session_title: "architect" } },
  as: "agent" as const,
};
const rows = [
  "Opening words.",
  "[lf](https://lf.test/?q=a&#10;b) one",
  "[tab](https://tab.test/a&#9;b) two",
  "[ff](https://ff.test/a&#12;b) three",
  "[del](https://del.test/a&#127;b) four",
  "&lt;javascript:alert(1);'a@js.cc'> five",
  "&lt;mailto:x;a@mt.cc> six",
  "&lt;javascript:void(0);www.void.test> seven",
  "x &lt;a:b;https://y1.test> y &lt;c:d;https://z1.test> eight",
  "Closing words.",
];

test.beforeEach(async () => {
  await resetDatabase();
});

test("round-3 probe", async ({ browser }) => {
  test.setTimeout(120_000);
  await createProject({ key: "CORE", name: "Core" });
  // Each row alone first: refused or stored, and as what.
  for (const [index, row] of rows.entries()) {
    try {
      const one = await createIssue({ project: "CORE", spec: `${row}\n`, title: `zebra${index} quokka${index}` });
      console.log(`PROBE alone ${JSON.stringify(row)} -> ${JSON.stringify((await getArtifactText(one.primary_artifact_id)).markdown)}`);
    } catch (error) {
      console.log(`PROBE alone ${JSON.stringify(row)} -> REFUSED ${String(error).slice(0, 300)}`);
    }
  }
  const spec = `${rows.join("\n\n")}\n`;
  const issue = await createIssue({ project: "CORE", spec, title: "Round three probe" });
  await createComment(issue.key, { body: spec });
  const id = issue.primary_artifact_id;
  const context = await asUser(browser, "alice");
  const page = await context.newPage();
  await page.goto(`/issues/${issue.key}/conversation`);
  const turns = page.getByRole("list", { name: "Conversation turns" });
  await expect(turns).toContainText("Closing words.");
  for (const link of await turns.getByRole("link").all()) {
    const href = await link.getAttribute("href");
    if (href && /test|cc|javascript|mailto/.test(href)) console.log(`PROBE comment link ${JSON.stringify(await link.textContent())} href ${JSON.stringify(href)}`);
  }
  const report = async (label: string) => {
    const artifact = await getArtifact(id);
    const latest = Math.max(...artifact.versions.map((v) => v.number));
    console.log(`PROBE ${label} text ${JSON.stringify((await getArtifactText(id)).markdown)}`);
    console.log(`PROBE ${label} stored v${latest} ${JSON.stringify((await getArtifactVersion(id, latest)).markdown)}`);
    await openSpec(page, issue.key, "Closing words");
    for (const link of await documentEditor(page).getByRole("link").all()) {
      console.log(`PROBE ${label} editor link ${JSON.stringify(await link.textContent())} href ${JSON.stringify(await link.getAttribute("href"))}`);
    }
  };
  await report("write");
  await editArtifact(id, { ops: [{ op: "replace", find: "Closing words.", with: "Closing words, edited." }] }, session);
  await expect.poll(async () => (await getArtifactText(id)).markdown).toContain("edited.");
  await report("edit-1");
  await editArtifact(id, { ops: [{ op: "replace", find: "Opening words.", with: "Opening words, edited." }] }, session);
  await expect.poll(async () => (await getArtifactText(id)).markdown).toContain("Opening words, edited.");
  await report("edit-2");
  await context.close();
});
