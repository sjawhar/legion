// dispatchpub publishes Dispatch-shaped envelopes straight to the Envoy notification stream, at a
// steady rate, on the issue subjects the load test's sessions follow
// (notifications.dispatch.issue.<KEY>.comment.created). Each envelope has the shape Dispatch's
// outbox gives one (packages/envoy/internal/dispatch/outbox/publisher.go envelope()): a
// contracts.Envelope whose payload is a JSON model.Event of type comment.created, published to
// JetStream under Nats-Msg-Id "<dedupe key>:<topic>", as bus.publishJetStream does.
//
// It is the supplement of the INFRA-594 load re-check: beside a step-10x run, 40 envelopes a
// second for 600 s, each on an issue five of the step's sessions follow, so each one is relayed to
// five session consumers. The step's relay-frames.csv holds the frames; this tool's publishes.jsonl
// holds what was sent, so expected frames = published x 5.
package main

import (
	"context"
	"encoding/json"
	"flag"
	"fmt"
	"log"
	"os"
	"strings"
	"sync"
	"sync/atomic"
	"time"

	"github.com/nats-io/nats.go"
)

type seedIssue struct {
	Key string `json:"key"`
}

type seedFile struct {
	Issues []seedIssue `json:"issues"`
}

// envelope is contracts.Envelope's wire form, with the fields Dispatch's outbox sets.
type envelope struct {
	EventID        string `json:"event_id"`
	Source         string `json:"source"`
	SourceEventID  string `json:"source_event_id"`
	SourceSession  string `json:"source_session,omitempty"`
	Topic          string `json:"topic"`
	DedupeKey      string `json:"dedupe_key"`
	IssuedAt       int64  `json:"issued_at"`
	PayloadSummary string `json:"payload_summary"`
	Payload        string `json:"payload"`
	TraceID        string `json:"trace_id"`
}

// event is model.Event's wire form for a comment.created.
type event struct {
	ID        int64     `json:"id"`
	IssueKey  string    `json:"issue_key"`
	Project   string    `json:"project"`
	Seq       int       `json:"seq"`
	Type      string    `json:"type"`
	Actor     actor     `json:"actor"`
	Notify    bool      `json:"notify"`
	CreatedAt time.Time `json:"created_at"`
	Payload   any       `json:"payload"`
}

type actor struct {
	Kind string `json:"kind"`
	ID   string `json:"id"`
}

type record struct {
	N        int    `json:"n"`
	EventID  int64  `json:"event_id"`
	Topic    string `json:"topic"`
	SentNs   int64  `json:"sent_ns"`
	AckSeq   uint64 `json:"ack_seq"`
	AckNs    int64  `json:"ack_ns"`
	Dup      bool   `json:"duplicate"`
	Err      string `json:"error,omitempty"`
	IssueKey string `json:"issue_key"`
}

func main() {
	url := flag.String("nats", "nats://127.0.0.1:14222", "the stack's NATS URL")
	seedPath := flag.String("seed", "", "the stack's seed.json (required)")
	issues := flag.Int("issues", 1000, "use the first N issues in key order, the step's active issues")
	rate := flag.Float64("rate", 40, "envelopes per second")
	duration := flag.Duration("duration", 600*time.Second, "how long to publish")
	idBase := flag.Int64("id-base", 9_000_000_000, "the first event id; far above any id the step's writes get")
	tag := flag.String("tag", "sup", "a tag in each comment body and dedupe key, so one run's envelopes are its own")
	out := flag.String("out", "publishes.jsonl", "where to write one record per publish")
	flag.Parse()
	if *seedPath == "" {
		log.Fatal("dispatchpub: -seed is required")
	}
	data, err := os.ReadFile(*seedPath)
	if err != nil {
		log.Fatalf("dispatchpub: read seed: %v", err)
	}
	var sf seedFile
	if err := json.Unmarshal(data, &sf); err != nil {
		log.Fatalf("dispatchpub: decode seed: %v", err)
	}
	if len(sf.Issues) < *issues {
		log.Fatalf("dispatchpub: the seed has %d issues, fewer than the %d asked for", len(sf.Issues), *issues)
	}
	keys := make([]string, *issues)
	for i := range keys {
		keys[i] = sf.Issues[i].Key
	}

	nc, err := nats.Connect(*url, nats.Name("dispatchpub"), nats.MaxReconnects(-1))
	if err != nil {
		log.Fatalf("dispatchpub: connect: %v", err)
	}
	defer nc.Close()
	js, err := nc.JetStream(nats.PublishAsyncMaxPending(4096))
	if err != nil {
		log.Fatalf("dispatchpub: jetstream: %v", err)
	}
	f, err := os.Create(*out)
	if err != nil {
		log.Fatalf("dispatchpub: create %s: %v", *out, err)
	}
	defer f.Close()
	enc := json.NewEncoder(f)

	// body is the comment text: the tag, the event's number and filler, so an envelope is the size
	// a seeded comment's is (a few hundred bytes).
	filler := strings.Repeat("the quick brown fox jumps over the lazy dog. ", 5)

	var sent, acked, failed, dups atomic.Int64
	var acks sync.WaitGroup
	ctx, cancel := context.WithCancel(context.Background())
	defer cancel()
	records := make(chan record, 1<<16)
	done := make(chan struct{})
	go func() {
		defer close(done)
		for r := range records {
			_ = enc.Encode(r)
		}
	}()

	interval := time.Duration(float64(time.Second) / *rate)
	ticker := time.NewTicker(interval)
	defer ticker.Stop()
	deadline := time.Now().Add(*duration)
	start := time.Now()
	n := 0
	for now := range ticker.C {
		if now.After(deadline) {
			break
		}
		key := keys[n%len(keys)]
		id := *idBase + int64(n)
		eventID := fmt.Sprintf("dispatch-%s-%d", *tag, id)
		topic := "notifications.dispatch.issue." + key + ".comment.created"
		ev := event{ID: id, IssueKey: key, Project: key[:strings.IndexByte(key, '-')], Seq: n + 1,
			Type: "comment.created", Actor: actor{Kind: "user", ID: "person-0"}, Notify: true, CreatedAt: now.UTC(),
			Payload: map[string]any{
				"id":   fmt.Sprintf("cmt_%s_%d", *tag, id),
				"body": fmt.Sprintf("%s supplement %d on %s. %s", *tag, n, key, filler),
			}}
		payload, err := json.Marshal(ev)
		if err != nil {
			log.Fatalf("dispatchpub: encode event: %v", err)
		}
		item := envelope{EventID: eventID, Source: "dispatch", SourceEventID: fmt.Sprint(id), Topic: topic,
			DedupeKey: eventID, IssuedAt: now.UnixMilli(),
			PayloadSummary: key + " comment created", Payload: string(payload), TraceID: eventID}
		body, err := json.Marshal(item)
		if err != nil {
			log.Fatalf("dispatchpub: encode envelope: %v", err)
		}
		sentNs := time.Now().UnixNano()
		future, err := js.PublishMsgAsync(&nats.Msg{Subject: topic, Data: body}, nats.MsgId(eventID+":"+topic))
		if err != nil {
			failed.Add(1)
			records <- record{N: n, EventID: id, Topic: topic, SentNs: sentNs, Err: err.Error(), IssueKey: key}
			n++
			continue
		}
		sent.Add(1)
		acks.Add(1)
		go func(n int, id int64, topic, key string, sentNs int64, future nats.PubAckFuture) {
			defer acks.Done()
			r := record{N: n, EventID: id, Topic: topic, SentNs: sentNs, IssueKey: key}
			select {
			case ack := <-future.Ok():
				r.AckSeq, r.Dup, r.AckNs = ack.Sequence, ack.Duplicate, time.Now().UnixNano()
				acked.Add(1)
				if ack.Duplicate {
					dups.Add(1)
				}
			case err := <-future.Err():
				r.Err = err.Error()
				failed.Add(1)
			case <-ctx.Done():
				r.Err = "no acknowledgement before the end of the run"
				failed.Add(1)
			}
			records <- r
		}(n, id, topic, key, sentNs, future)
		n++
	}
	// Give the acknowledgements of the last publishes their time, then stop the writer.
	select {
	case <-js.PublishAsyncComplete():
	case <-time.After(30 * time.Second):
	}
	cancel()
	acks.Wait()
	close(records)
	<-done
	log.Printf("dispatchpub: %d publishes in %.1fs (%.2f/s): %d acknowledged, %d duplicates, %d failed; records in %s",
		n, time.Since(start).Seconds(), float64(n)/time.Since(start).Seconds(), acked.Load(), dups.Load(), failed.Load(), *out)
}
