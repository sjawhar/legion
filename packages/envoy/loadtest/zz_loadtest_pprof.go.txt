// zz_loadtest_pprof.go: added by packages/envoy/loadtest/stack.sh (build_dispatch_at) to a
// Dispatch tree that has no cmd/dispatch/pprof.go of its own, so a build of an older commit can be
// profiled like the checkout's. Not part of that commit: stack.sh records it as
// dispatch_pprof_overlay in the binary's .json.
//
// When LOADTEST_DISPATCH_PPROF_PORT is set, init binds 127.0.0.1:<port> before main runs (a value
// that is not a port 1-65535, or a bind that fails, exits 2) and serves net/http/pprof on a mux of
// its own, never http.DefaultServeMux, which Dispatch's own listener must not expose.
package main

import (
	"fmt"
	"net"
	"net/http"
	"net/http/pprof"
	"os"
	"strconv"
	"strings"
	"time"
)

func init() {
	raw := strings.TrimSpace(os.Getenv("LOADTEST_DISPATCH_PPROF_PORT"))
	if raw == "" {
		return
	}
	port, err := strconv.Atoi(raw)
	if err != nil || port < 1 || port > 65535 {
		fmt.Fprintf(os.Stderr, "LOADTEST_DISPATCH_PPROF_PORT: %q is not a port 1-65535\n", raw)
		os.Exit(2)
	}
	addr := net.JoinHostPort("127.0.0.1", strconv.Itoa(port))
	listener, err := net.Listen("tcp", addr)
	if err != nil {
		fmt.Fprintf(os.Stderr, "LOADTEST_DISPATCH_PPROF_PORT: listen on %s: %v\n", addr, err)
		os.Exit(2)
	}
	mux := http.NewServeMux()
	mux.HandleFunc("/debug/pprof/", pprof.Index)
	mux.HandleFunc("/debug/pprof/cmdline", pprof.Cmdline)
	mux.HandleFunc("/debug/pprof/profile", pprof.Profile)
	mux.HandleFunc("/debug/pprof/symbol", pprof.Symbol)
	mux.HandleFunc("/debug/pprof/trace", pprof.Trace)
	for _, name := range []string{"allocs", "block", "goroutine", "heap", "mutex", "threadcreate"} {
		mux.Handle("/debug/pprof/"+name, pprof.Handler(name))
	}
	server := &http.Server{Handler: mux, ReadHeaderTimeout: 10 * time.Second}
	go func() {
		if err := server.Serve(listener); err != nil && err != http.ErrServerClosed {
			fmt.Fprintf(os.Stderr, "LOADTEST_DISPATCH_PPROF_PORT: pprof listener on %s stopped: %v\n", addr, err)
		}
	}()
	fmt.Fprintf(os.Stderr, "loadtest pprof listening on %s\n", listener.Addr())
}
